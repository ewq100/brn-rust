import argparse, datetime, hashlib, importlib.util, json, time
from pathlib import Path
ROOT=Path('EVIDENCE')
REPO=Path('SOURCE_CHECKOUT')
SPEC=importlib.util.spec_from_file_location('guidance', REPO/'scripts/evaluate-wiki-guidance.py')
g=importlib.util.module_from_spec(SPEC);SPEC.loader.exec_module(g)
PRIOR=Path('QUALIFIED_HISTORICAL_HELPER_SOURCE')
IDS=['meeting-ambiguous-discussion','meeting-clarification-unknown-owner','meeting-duplicate-notes-one-followup','meeting-changed-commitment-conflict']
all_cases=json.loads(g.CASES.read_text())['cases']; cases=[c for identity in IDS for c in all_cases if c['id']==identity]
assert [c['id'] for c in cases]==IDS and sum(len(c['steps']) for c in cases)==6
case_id=argparse.ArgumentParser();case_id.add_argument('case',choices=[IDS[2]]);chosen=case_id.parse_args().case
case=next(c for c in cases if c['id']==chosen)
record_path=ROOT/'record.json'
def read():return json.loads(record_path.read_text())
def write(r):record_path.write_text(json.dumps(r,ensure_ascii=False,indent=2)+'\n')
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
r=read(); assert not r.get('stop') and not r.get('closed')
assert r['limits_reconciliation']['owner_tool_round_limit']==12
assert hashlib.sha256((REPO/'scripts/evaluate-wiki-guidance.py').read_bytes()).hexdigest()==r['script_hashes']['scripts/evaluate-wiki-guidance.py']
root=ROOT/'meetings'
if not root.exists():g.create_root(root)
assert (root/'EVALUATION-OWNED').is_file() and root.resolve().is_relative_to(ROOT)
folder=root/chosen
assert not folder.exists(), 'No repeated case allowed'
original=g.Cli.call
attempts=[]
def observed(self,command,**options):
 live=command=='ask' or (command=='intake' and options.get('intent')=='useful')
 a=None
 if command=='configure':options['helper']='off'
 if live:
  options['trace']='full'
  current=read(); assert not current.get('stop');assert len(current['attempts'])<2
  current.setdefault('continuation_deadline_monotonic',time.monotonic()+15*60)
  current.setdefault('live_started_at_utc',now())
  assert current['continuation_deadline_monotonic']-time.monotonic()>=180, 'Insufficient live-time bound for a 180-second command'
  a={'ordinal':len(current['attempts'])+1,'case':chosen,'command':command,'reserved_at_utc':now(),'status':'reserved_before_launch','timeout_seconds':180,'tool_round_budget':options.get('budget',12),'model':current['model'],'effort':'medium'}
  assert a['tool_round_budget']<=12
  current['attempts'].append(a); write(current); attempts.append(a['ordinal'])
 safe_options={k:('EXISTING_AUTH_FILE' if k=='auth_file' else str(v)) for k,v in options.items()}
 timestamp=now()
 try:
  out=original(self,command,**options)
 except BaseException as e:
  current=read();current['stop']={'case':chosen,'reason':str(e),'timestamp_utc':now()}
  if a:current['attempts'][a['ordinal']-1]['status']='failed_or_uncertain'
  write(current);raise
 with (folder/'commands-redacted.jsonl').open('a') as f:
  f.write(json.dumps({'timestamp_utc':timestamp,'binary':str(self.binary),'data_dir':str(self.data),'command':command,'options':safe_options,'observation':self.commands[-1]},ensure_ascii=False)+'\n')
 if a:
  current=read();current['attempts'][a['ordinal']-1].update(status='returned',terminal=out.get('terminal'),usage=out.get('trace',{}).get('usage','unavailable'),model_turns=out.get('trace',{}).get('model_turns','unavailable'),tool_rounds=out.get('trace',{}).get('rounds','unavailable'));write(current)
 if a:
  # Only observation reads here: no Apply, model, helper Run or changed oracle.
  step=len(attempts)
  safe=g.public_snapshot(g.snapshot(self))
  before=json.loads((folder/'result.json').read_text())['before']
  # First intake has only update-0. Same unchanged source oracle on the
  # observed prefix; the full unmodified fixture/oracle runs after intake 2.
  if step==1:
   checks={'protected_references_preserved':g.protection_preserved(before['records'],safe['records'])}
   checks.update(g.action_checks(case['action_expectation'],safe))
   checks.update(g.action_source_checks([0],safe))
  else:
   checks=g.structural_checks(case,before,safe)
  checks['no_accepted_action_or_note']=not any(any(k in record['data'] for k in ['Action','Note']) and not record['archived'] for record in safe['records'])
  checks['no_applied_receipt']=not safe['receipts']
  action_candidates=[candidate for candidate in safe['candidates'] if any(not w['archived'] and 'Action' in w['data'] for w in candidate['request']['writes'])]
  checks['one_active_pending_action_candidate']=len(action_candidates)==1 and sum(1 for c in action_candidates for w in c['request']['writes'] if not w['archived'] and 'Action' in w['data'])==1
  reopened=g.public_snapshot(g.snapshot(self))
  checks['fresh_process_snapshot_exact']=reopened==safe
  (folder/f'step-{step}-fresh-process.json').write_text(json.dumps(reopened,ensure_ascii=False,indent=2)+'\n')
  predecessor=None
  if step==2:
   previous=json.loads((folder/'step-1-inspection.json').read_text())['snapshot']['candidates'][0]['operation']
   predecessor=g.public_snapshot(self.envelope('operation',id=previous))
   (folder/'predecessor-after-step-2.json').write_text(json.dumps(predecessor,ensure_ascii=False,indent=2)+'\n')
   active=action_candidates[0]['operation'] if len(action_candidates)==1 else None
   successor=g.public_snapshot(self.envelope('operation',id=active)) if active else None
   (folder/'successor-after-step-2.json').write_text(json.dumps(successor,ensure_ascii=False,indent=2)+'\n')
   checks['replacement_predecessor_retired']=active!=previous and predecessor.get('result',{}).get('state')=='retired'
   checks['replacement_tracks_predecessor']=bool(successor and successor.get('result',{}).get('replacement_of')==previous)
   checks['replacement_still_prepared']=bool(successor and successor.get('result',{}).get('state')=='prepared')


  failures=[name for name,passed in checks.items() if passed is not True]
  if out.get('terminal')!='Completed':failures.append('terminal_not_Completed')
  receipt={'predecessor':predecessor,'step':step,'outcome':g.public_snapshot(out),'snapshot':safe,'checks':checks,'failures':failures,'observed_at_utc':now()}
  (folder/f'step-{step}-inspection.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n')
  if failures:
   current=read();current['stop']={'case':chosen,'step':step,'reason':', '.join(failures),'timestamp_utc':now()};write(current)
   raise g.CallFailed('Stop gate: '+', '.join(failures))
  print(json.dumps({'awaiting_actual_byte_and_prose_review':str(folder/f'step-{step}-inspection.json')}),flush=True)
  approval=folder/f'step-{step}-review.json'
  while not approval.exists():
   if time.monotonic()>=read()['continuation_deadline_monotonic']:
    current=read();current['stop']={'reason':'15 minute live limit reached during review','timestamp_utc':now()};write(current);raise g.CallFailed('15 minute bound')
   time.sleep(.5)
  decision=json.loads(approval.read_text())
  if decision.get('verdict')!='pass':
   current=read();current['stop']={'case':chosen,'step':step,'reason':decision.get('reason','Actual semantic review failed'),'timestamp_utc':now()};write(current);raise g.CallFailed('Actual semantic review failed')
 return out
g.Cli.call=observed
args=argparse.Namespace(live=True,provider='Codex',auth_file=Path('EXISTING_AUTH_FILE'),credentials_dir=None,helper=None,stock_helper=ROOT/'bin/brn-xberg-helper',brn_helper=ROOT/'bin/brn-xberg-helper-brn',timeout=180,filler=0)
selection={'provider':'Codex','model':r['model'],'effort':'medium'}
result=g.run_arm(ROOT/'bin/brn',root,folder,case,selection,args)
current=read();current.setdefault('meeting_results',{})[chosen]={'completed':result['completed'],'failure':result['failure'],'verification_failure':result['verification_failure'],'semantic_status':result['semantic_status'],'result':str(folder/'result.json')}
if not result['completed'] or result['failure'] or result['verification_failure']:
 current['stop']={'case':chosen,'reason':result['failure'] or result['verification_failure'] or 'Not completed','timestamp_utc':now()}
write(current)
print(json.dumps(current['meeting_results'][chosen],ensure_ascii=False))
raise SystemExit(0 if result['completed'] else 1)
